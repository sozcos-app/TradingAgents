## 一、后端任务

### 1.1 创建数据模型 `app/models/calendar.py`

定义以下 Pydantic 模型（参照 `app/models/user.py` 的 PyObjectId 模式）：

```
MacroIndicator
├── id: Optional[PyObjectId]
├── indicator_name: str          # 如 "PMI", "CPI", "社融同比"
├── value: float                 # 指标值
├── unit: str                    # 单位，如 "%", "亿"
├── date: datetime               # 数据日期
├── source: str                  # 数据来源，如 "AKShare", "Tushare"
├── year_on_year: Optional[float]  # 同比变化
├── month_on_month: Optional[float] # 环比变化
├── score: Optional[int]         # 单项评分(0~100)
├── created_at: datetime
└── updated_at: datetime

MacroScore
├── id: Optional[PyObjectId]
├── total_score: float           # 总分 0~100
├── level: str                   # "可积极参与" / "谨慎试仓" / "防守观望"
├── details: dict                # 各指标得分明细
├── date: datetime
└── created_at: datetime

IndustryProsperity
├── id: Optional[PyObjectId]
├── industry_name: str
├── inventory_cycle: str         # 库存周期阶段
├── profit_trend: str            # 盈利趋势
├── demand_growth: str           # 需求增速
├── capital_flow: str            # 资金流向
├── policy_support: str          # 政策支持
├── score: float                 # 景气评分 0~100
├── suggestion: str              # "超配" / "标配" / "低配"
├── date: datetime
└── created_at: datetime

StockHealthCheck
├── id: Optional[PyObjectId]
├── ts_code: str                 # 股票代码
├── stock_name: str
├── main_business_score: float   # 主营匹配度 0~30
├── profit_quality_score: float  # 利润含金量 0~25
├── gross_margin_score: float    # 毛利率合理性 0~20
├── disclosure_score: float      # 信披记录 0~15
├── supply_chain_score: float    # 供应链验证 0~10
├── total_score: float           # 总分 0~100
├── risk_level: str              # "低" / "中" / "高"
├── conclusion: str              # "可关注" / "谨慎" / "回避"
├── alerts: list                 # 预警项列表
├── date: datetime
└── created_at: datetime

MarketEvent
├── id: Optional[PyObjectId]
├── title: str                   # 事件标题
├── event_type: str              # "财报" / "政策" / "解禁" / "地缘" / "指数调整" 等
├── event_date: datetime         # 事件日期
├── impact_direction: str        # "利多" / "利空" / "中性"
├── impact_strength: str         # "高" / "中" / "低"
├── affected_sectors: list[str]  # 影响板块
├── action_suggestion: str       # 行动建议
├── source: str                  # 信息来源
├── description: str             # 事件描述
├── is_auto: bool                # 是否自动采集
├── created_by: str              # 创建人
├── created_at: datetime
└── updated_at: datetime
```

### 1.3 创建服务层 `app/services/calendar_service.py`

创建 4 个 Service 类（先写空壳方法，后续 Phase 填充逻辑）：

```python
class MacroDataService:
    """宏观感知服务"""
    async def fetch_indicators(self, names: list[str]) -> list    # Phase 2 实现
    async def get_indicator_history(self, name, start, end) -> list
    async def get_latest_indicators(self) -> list
    async def calculate_score(self) -> MacroScore                 # Phase 3 实现

class IndustryProsperityService:
    """行业景气服务"""
    async def fetch_industry_data(self, industry: str) -> dict    # Phase 5 实现
    async def get_ranking(self) -> list
    async def get_detail(self, industry: str) -> dict
    async def calculate_score(self, industry: str) -> dict

class StockHealthService:
    """个股体检服务"""
    async def check_stock(self, ts_code: str) -> dict             # Phase 6 实现
    async def get_report(self, ts_code: str) -> dict
    async def get_alerts(self, ts_code: str) -> list

class MarketEventService:
    """市场事件服务"""
    async def get_events(self, query: EventQuery) -> list         # Phase 7 实现
    async def create_event(self, data: EventCreate) -> str
    async def update_event(self, id: str, data: dict) -> bool
    async def delete_event(self, id: str) -> bool
    async def get_upcoming(self, days: int = 7) -> list

# 全局实例
macro_service = MacroDataService()
industry_service = IndustryProsperityService()
stock_health_service = StockHealthService()
event_service = MarketEventService()
```

### 1.4 创建路由 `app/routers/calendar.py`

```
POST   /api/calendar/macro/fetch              # 手动触发数据采集
GET    /api/calendar/macro/indicators          # 查询宏观指标列表
GET    /api/calendar/macro/indicators/{name}   # 单个指标历史
POST   /api/calendar/macro/score               # 触发评分计算
GET    /api/calendar/macro/score/latest         # 获取最新评分

GET    /api/calendar/industry/ranking           # 行业景气排名
GET    /api/calendar/industry/{name}/detail     # 单行业详情
POST   /api/calendar/industry/{name}/score      # 触发行业评分
GET    /api/calendar/industry/{name}/history     # 行业评分历史

POST   /api/calendar/stock-health/check         # 触发个股体检
GET    /api/calendar/stock-health/{ts_code}      # 获取体检报告
GET    /api/calendar/stock-health/{ts_code}/alerts  # 预警信息

GET    /api/calendar/events                      # 事件列表(分页+筛选)
GET    /api/calendar/events/upcoming             # 即将到来事件
POST   /api/calendar/events                      # 创建事件(手动)
PUT    /api/calendar/events/{id}                 # 更新事件
DELETE /api/calendar/events/{id}                 # 删除事件
GET    /api/calendar/events/types                 # 事件类型字典
```

### 1.5 注册路由到 `app/main.py`

在现有路由注册区域追加：


### 1.7 创建 API 模块 `frontend/src/api/calendar.ts`

```typescript
export const calendarApi = {
  // 宏观
  macro: {
    fetch: () => ApiClient.post('/api/calendar/macro/fetch'),
    getIndicators: () => ApiClient.get('/api/calendar/macro/indicators'),
    getIndicatorHistory: (name: string) => ApiClient.get(`/api/calendar/macro/indicators/${name}`),
    triggerScore: () => ApiClient.post('/api/calendar/macro/score'),
    getLatestScore: () => ApiClient.get('/api/calendar/macro/score/latest'),
  },
  // 行业
  industry: {
    getRanking: () => ApiClient.get('/api/calendar/industry/ranking'),
    getDetail: (name: string) => ApiClient.get(`/api/calendar/industry/${name}/detail`),
  },
  // 个股
  stockHealth: {
    check: (tsCode: string) => ApiClient.post('/api/calendar/stock-health/check', { ts_code: tsCode }),
    getReport: (tsCode: string) => ApiClient.get(`/api/calendar/stock-health/${tsCode}`),
  },
  // 事件
  events: {
    list: (params: EventQuery) => ApiClient.get('/api/calendar/events', { params }),
    getUpcoming: () => ApiClient.get('/api/calendar/events/upcoming'),
    create: (data: any) => ApiClient.post('/api/calendar/events', data),
    update: (id: string, data: any) => ApiClient.put(`/api/calendar/events/${id}`, data),
    delete: (id: string) => ApiClient.delete(`/api/calendar/events/${id}`),
    getTypes: () => ApiClient.get('/api/calendar/events/types'),
  },
}
```
